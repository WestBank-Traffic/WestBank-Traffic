document.getElementById('get-directions-btn').addEventListener('click', function() { const destination = document.getElementById('destination-input').value.trim(); if (!destination) { alert("الرجاء إدخال الوجهة."); return; } if (!userCoordinates) { alert("الرجاء تحديد موقعك الحالي أولاً باستخدام زر الموقع."); return; } // إزالة التحكم في التوجيه الحالي إذا كان موجوداً if (routingControl) { map.removeControl(routingControl); routingControl = null; } // إزالة علامة نقطة النهاية السابقة إذا كانت موجودة if (destinationMarker) { map.removeLayer(destinationMarker); destinationMarker = null; } var geocoder = L.Control.Geocoder.nominatim(); // البحث عن الوجهة باستخدام Geocoder geocoder.geocode(destination, function(results) { if (results && results.length > 0) { const destinationLatLng = results[0].center; // إضافة علامة في الوجهة المحددة بواسطة Geocoder destinationMarker = L.marker(destinationLatLng).addTo(map).bindPopup(`نقطة النهاية: ${results[0].name}`).openPopup(); map.setView(destinationLatLng, 14); // يمكنك تعديل مستوى التكبير هنا // إنشاء المسار بين الموقع الحالي للمستخدم والوجهة المدخلة routingControl = L.Routing.control({ waypoints: [ L.latLng(userCoordinates[0], userCoordinates[1]), // الموقع الحالي للمستخدم destinationLatLng // الوجهة ], router: L.Routing.osrmv1({ serviceUrl: 'https://router.project-osrm.org/route/v1' // خدمة التوجيه }), routeWhileDragging: true, language: 'ar', showAlternatives: false, lineOptions: { styles: [{color: 'red', weight: 6, opacity: 0.8}] // تعيين لون المسار إلى الأحمر }, collapsible: true }).addTo(map); routingControl.on('routesfound', function(e) { var routes = e.routes; var summary = routes[0].summary; alert('المسافة: ' + (summary.totalDistance / 1000).toFixed(2) + ' كم، المدة: ' + Math.round(summary.totalTime / 60) + ' دقيقة.'); }); routingControl.on('routingerror', function(e) { alert('لم يتم العثور على مسار للوجهة المحددة. يرجى التحقق من الوجهة أو المحاولة مرة أخرى.'); console.error("Routing error:", e.error); }); } else { alert('لم يتم العثور على الوجهة المحددة: ' + destination + '. الرجاء التحقق من الإدخال.'); } }); });
